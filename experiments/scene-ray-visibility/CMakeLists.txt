cmake_minimum_required(VERSION 3.20)
project(SceneRayVisibility LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(root ${repo}/build/scene-ray-visibility)
set(TINYBVH_ROOT "$ENV{HOME}/Git/tinybvh" CACHE PATH "Pinned upstream research dependency")
add_subdirectory(${root}/source/libsoftgl candidate-library)
add_subdirectory(${root}/baseline-source/libsoftgl baseline-library)
file(READ ${root}/source/backend.txt backend)
string(STRIP "${backend}" backend)
if(backend STREQUAL "scalar")
    enable_language(CXX)
    add_library(ray_bvh STATIC bridge.cpp)
    target_include_directories(ray_bvh PRIVATE ${TINYBVH_ROOT})
    if(EMSCRIPTEN)
        target_compile_definitions(ray_bvh PRIVATE TINYBVH_NO_SIMD)
    else()
        target_compile_options(ray_bvh PRIVATE -mavx2 -mfma -msse4.2)
    endif()
else()
    add_library(ray_bvh STATIC bvh.c)
    set(SCENE_BVH_LEAF_SIZE 4 CACHE STRING "Maximum primitive count before SAH splitting")
    if(NOT SCENE_BVH_LEAF_SIZE MATCHES "^[0-9]+$" OR SCENE_BVH_LEAF_SIZE LESS 1 OR SCENE_BVH_LEAF_SIZE GREATER 256)
        message(FATAL_ERROR "SCENE_BVH_LEAF_SIZE must be between 1 and 256")
    endif()
    target_compile_definitions(ray_bvh PRIVATE SCENE_BVH_LEAF_SIZE=${SCENE_BVH_LEAF_SIZE})
endif()
target_link_libraries(softgl PUBLIC ray_bvh)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(source ${root}/baseline-source)
        set(library baseline_softgl)
    else()
        set(source ${root}/source)
        set(library softgl)
    endif()
    add_executable(${variant} ${repo}/experiments/glimpsw-mesa-comparison/softgl_bmw.c ${source}/model_wrap.c)
    add_executable(quality_${variant} ${repo}/experiments/scene-hierarchical-depth/quality_frames.c ${source}/model_wrap.c)
    add_executable(resident_${variant} ${repo}/experiments/scene-material-visibility/resident_trial.c ${source}/model_wrap.c)
    foreach(target ${variant} quality_${variant} resident_${variant})
        target_compile_definitions(${target} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY SOFTGL_MODEL_SCENE_POSITIONS)
        if(variant STREQUAL "candidate")
            target_compile_definitions(${target} PRIVATE SOFTGL_MODEL_SCENE_RAYS)
        endif()
        target_include_directories(${target} PRIVATE ${source}/libsoftgl/src)
        target_compile_options(${target} PRIVATE -msse4.1)
        target_link_libraries(${target} PRIVATE ${library} pthread m)
    endforeach()
endforeach()
add_executable(bvh_contract scene_bvh_contract.c)
target_include_directories(bvh_contract PRIVATE ${root}/source/libsoftgl/src)
target_compile_options(bvh_contract PRIVATE -msse4.1)
target_link_libraries(bvh_contract PRIVATE softgl pthread m)
option(SOFTGL_RAY_PROFILE "Build a diagnostic gperftools driver" OFF)
if(SOFTGL_RAY_PROFILE)
    find_path(PROFILER_INCLUDE_DIR gperftools/profiler.h REQUIRED)
    find_library(PROFILER_LIBRARY profiler REQUIRED)
    add_executable(profile_candidate ${repo}/experiments/native-cpu-profiles/profile_scene.c ${root}/source/model_wrap.c)
    target_include_directories(profile_candidate PRIVATE ${root}/source/libsoftgl/src ${PROFILER_INCLUDE_DIR})
    target_compile_definitions(profile_candidate PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_SCENE_RAYS)
    target_compile_options(profile_candidate PRIVATE -msse4.1)
    target_link_libraries(profile_candidate PRIVATE softgl ${PROFILER_LIBRARY} pthread m)
endif()
