cmake_minimum_required(VERSION 3.20)
project(SceneMaterialVisibility LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(source ${repo}/build/scene-material-visibility/source)
add_subdirectory(${source}/libsoftgl libsoftgl)
add_executable(candidate ${repo}/experiments/glimpsw-mesa-comparison/softgl_bmw.c ${source}/model_wrap.c)
target_compile_definitions(candidate PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY)
target_include_directories(candidate PRIVATE ${source}/libsoftgl/src)
target_compile_options(candidate PRIVATE -msse4.1)
target_link_libraries(candidate PRIVATE softgl pthread m)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(quality_source ${repo}/build/scene-material-visibility/baseline-source)
        set(quality_library ${repo}/build/fused-material-pass/native/libsoftgl/libsoftgl.a)
    else()
        set(quality_source ${source})
        set(quality_library softgl)
    endif()
    add_executable(quality_${variant} ${repo}/experiments/fused-material-pass/quality_frames.c ${quality_source}/model_wrap.c)
    target_compile_definitions(quality_${variant} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES)
    if(variant STREQUAL "candidate")
        target_compile_definitions(quality_${variant} PRIVATE SOFTGL_MODEL_SCENE_VISIBILITY)
    endif()
    target_include_directories(quality_${variant} PRIVATE ${quality_source}/libsoftgl/src ${quality_source}/libsoftgl/include)
    target_compile_options(quality_${variant} PRIVATE -msse4.1)
    target_link_libraries(quality_${variant} PRIVATE ${quality_library} pthread m)
endforeach()
option(SCENE_PROFILE "Build the diagnostic CPU profiler" OFF)
if(SCENE_PROFILE)
    find_path(PROFILER_INCLUDE_DIR gperftools/profiler.h REQUIRED)
    find_library(PROFILER_LIBRARY profiler REQUIRED)
    add_executable(profile_scene ${repo}/experiments/native-cpu-profiles/profile_scene.c ${source}/model_wrap.c)
    target_include_directories(profile_scene PRIVATE ${source}/libsoftgl/src ${PROFILER_INCLUDE_DIR})
    target_compile_definitions(profile_scene PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY)
    target_compile_options(profile_scene PRIVATE -msse4.1 -g -fno-omit-frame-pointer)
    target_link_libraries(profile_scene PRIVATE softgl ${PROFILER_LIBRARY} pthread m)
endif()
add_executable(scene_contract contract.c)
target_include_directories(scene_contract PRIVATE ${source}/libsoftgl/src)
target_compile_options(scene_contract PRIVATE -msse4.1)
target_link_libraries(scene_contract PRIVATE softgl pthread m)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(resident_source ${repo}/build/scene-material-visibility/baseline-source)
        set(resident_library ${repo}/build/fused-material-pass/native/libsoftgl/libsoftgl.a)
    else()
        set(resident_source ${source})
        set(resident_library softgl)
    endif()
    add_executable(resident_${variant} resident_trial.c ${resident_source}/model_wrap.c)
    target_compile_definitions(resident_${variant} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES)
    if(variant STREQUAL "candidate")
        target_compile_definitions(resident_${variant} PRIVATE SOFTGL_MODEL_SCENE_VISIBILITY)
    endif()
    target_include_directories(resident_${variant} PRIVATE ${resident_source}/libsoftgl/src ${resident_source}/libsoftgl/include)
    target_compile_options(resident_${variant} PRIVATE -msse4.1)
    target_link_libraries(resident_${variant} PRIVATE ${resident_library} pthread m)
endforeach()
