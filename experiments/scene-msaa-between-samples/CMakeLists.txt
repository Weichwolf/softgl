cmake_minimum_required(VERSION 3.20)
project(SceneMsaaBetweenSamples LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only
    -msse4.1 -mno-avx -mno-avx2 -mno-avx512f)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCENE_TRIAL_ROOT "${repo}/build/scene-msaa-between-samples" CACHE PATH "Frozen experiment tree")
set(root ${SCENE_TRIAL_ROOT})
add_subdirectory(${root}/source/libsoftgl candidate-library)
add_subdirectory(${root}/baseline-source/libsoftgl baseline-library)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(source ${root}/baseline-source)
        set(library baseline_softgl)
    else()
        set(source ${root}/source)
        set(library softgl)
    endif()
    foreach(kind resident quality)
        if(kind STREQUAL "resident")
            set(driver ${repo}/experiments/scene-material-visibility/resident_trial.c)
        else()
            set(driver ${repo}/tools/experiment_support/quality_frames.c)
        endif()
        add_executable(${kind}_${variant} ${driver} ${source}/model_wrap.c)
        target_compile_definitions(${kind}_${variant} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
            SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
    foreach(kind quantized small)
        if(kind STREQUAL "quantized")
            set(fixture ${repo}/tests/scene_quantized.c)
        else()
            set(fixture ${repo}/experiments/scene-msaa-small-triangles/small_contract.c)
        endif()
        add_executable(${kind}_${variant} ${fixture})
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src ${repo}/tests)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
endforeach()
add_executable(hz_contract ${root}/source/hz_contract.c)
target_include_directories(hz_contract PRIVATE ${root}/source/libsoftgl/src ${repo}/tests)
target_link_libraries(hz_contract PRIVATE softgl pthread m)
option(SOFTGL_MSAA_VISIBILITY_AUDIT "Untimed real sample and between-sample counters" OFF)
if(SOFTGL_MSAA_VISIBILITY_AUDIT)
    target_compile_definitions(softgl PRIVATE SOFTGL_MSAA_VISIBILITY_AUDIT SOFTGL_MSAA_BETWEEN_AUDIT)
    add_executable(between_contract ${root}/source/between_contract.c)
    target_include_directories(between_contract PRIVATE ${root}/source/libsoftgl/src ${repo}/tests)
    target_link_libraries(between_contract PRIVATE softgl pthread m)
    add_executable(between_resident ${root}/source/between_resident.c ${root}/source/model_wrap.c)
    target_compile_definitions(between_resident PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
        SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
    target_include_directories(between_resident PRIVATE ${root}/source/libsoftgl/src)
    target_link_libraries(between_resident PRIVATE softgl pthread m)
    add_executable(msaa_contract ${root}/source/msaa_contract.c)
    target_include_directories(msaa_contract PRIVATE ${root}/source/libsoftgl/src ${root}/source ${repo}/tests)
    target_compile_definitions(msaa_contract PRIVATE SOFTGL_MSAA_ADAPTIVE_TEST)
    target_link_libraries(msaa_contract PRIVATE softgl pthread m)
endif()
