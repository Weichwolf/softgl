cmake_minimum_required(VERSION 3.30)
project(BmwComparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.3 GIT_SHALLOW TRUE)
FetchContent_Declare(meshoptimizer GIT_REPOSITORY https://github.com/zeux/meshoptimizer.git GIT_TAG v1.1 GIT_SHALLOW TRUE)
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 2c980bb59875b0d32144a71867fbdebb2f77cd20)
FetchContent_Declare(cgltf GIT_REPOSITORY https://github.com/jkuhlmann/cgltf.git GIT_TAG v1.15 GIT_SHALLOW TRUE)
FetchContent_Declare(tracy GIT_REPOSITORY https://github.com/wolfpld/tracy.git GIT_TAG v0.14.1 GIT_SHALLOW TRUE)
set(TRACY_ENABLE OFF CACHE BOOL "No profiler overhead in comparison" FORCE)
set(MESHOPT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MESHOPT_BUILD_GLTFPACK OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glm meshoptimizer stb cgltf tracy)
set(swr ${CMAKE_CURRENT_SOURCE_DIR}/GLimpSW/src/SwRast)
add_library(glimpsw STATIC ${swr}/Rasterizer.cpp ${swr}/ImageHelpers.cpp ${swr}/Scene.cpp ${swr}/Shading.cpp ${swr}/Voxel.cpp)
target_compile_options(glimpsw PUBLIC -march=native -mprefer-vector-width=512 -ffast-math -Wno-unused -Wsign-conversion -fno-stack-protector)
target_compile_definitions(glimpsw PUBLIC GLM_FORCE_XYZW_ONLY GLM_FORCE_DEPTH_ZERO_TO_ONE _CRT_SECURE_NO_WARNINGS)
target_include_directories(glimpsw PUBLIC ${swr} ${stb_SOURCE_DIR} ${cgltf_SOURCE_DIR} ${tracy_SOURCE_DIR}/public)
target_link_libraries(glimpsw PUBLIC glm::glm meshoptimizer TracyClient pthread)
target_precompile_headers(glimpsw PUBLIC <memory> <algorithm> <thread> <immintrin.h> <glm/glm.hpp>)
add_executable(glimpsw_bmw glimpsw_bmw.cpp)
target_link_libraries(glimpsw_bmw PRIVATE glimpsw)
add_subdirectory(../../libsoftgl libsoftgl)
target_compile_options(softgl PRIVATE -O3 -fno-strict-aliasing -ffast-math -fno-associative-math -fsigned-zeros -fno-finite-math-only)
add_executable(softgl_bmw softgl_bmw.c ../../wasm/model_wrap.c)
target_compile_definitions(softgl_bmw PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES)
target_include_directories(softgl_bmw PRIVATE ../../libsoftgl/include ../../libsoftgl/src)
target_compile_options(softgl_bmw PRIVATE -O3 -msse4.1 -fno-strict-aliasing -ffast-math -fno-associative-math -fsigned-zeros -fno-finite-math-only)
target_link_libraries(softgl_bmw PRIVATE softgl pthread m)
add_executable(mesa_bmw mesa_scene.c ../../wasm/model_wrap.c)
target_include_directories(mesa_bmw PRIVATE ../../libsoftgl/include)
target_compile_options(mesa_bmw PRIVATE -O3 -msse4.1 -fno-strict-aliasing -ffast-math -fno-associative-math -fsigned-zeros -fno-finite-math-only)
target_link_libraries(mesa_bmw PRIVATE OSMesa m)
