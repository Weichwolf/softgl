cmake_minimum_required(VERSION 3.20)
project(SceneSIMD128Policy LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only
    -msse4.1 -mno-avx -mno-avx2 -mno-avx512f)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(baseline ${repo}/build/scene-simd128-policy/baseline-source)
add_subdirectory(${repo}/libsoftgl candidate-library)
add_subdirectory(${baseline}/libsoftgl baseline-library)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(source ${baseline})
        set(library baseline_softgl)
        set(wrapper ${baseline}/model_wrap.c)
    else()
        set(source ${repo})
        set(library softgl)
        set(wrapper ${repo}/wasm/model_wrap.c)
    endif()
    add_executable(${variant} ${repo}/experiments/glimpsw-mesa-comparison/softgl_bmw.c ${wrapper})
    add_executable(quality_${variant} ${repo}/experiments/scene-hierarchical-depth/quality_frames.c ${wrapper})
    add_executable(resident_${variant} ${repo}/experiments/scene-material-visibility/resident_trial.c ${wrapper})
    foreach(target ${variant} quality_${variant} resident_${variant})
        target_compile_definitions(${target} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
            SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
        target_include_directories(${target} PRIVATE ${source}/libsoftgl/src)
        target_link_libraries(${target} PRIVATE ${library} pthread m)
    endforeach()
endforeach()
add_executable(simd128_contract ${repo}/tests/scene_native_wide.c)
target_include_directories(simd128_contract PRIVATE ${repo}/libsoftgl/src)
target_link_libraries(simd128_contract PRIVATE softgl pthread m)
