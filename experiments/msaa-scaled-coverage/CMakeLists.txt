cmake_minimum_required(VERSION 3.20)
project(MsaaScaledCoverage LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only
    -msse4.1 -mno-avx -mno-avx2 -mno-avx512f)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCENE_TRIAL_ROOT "${repo}/build/msaa-scaled-coverage" CACHE PATH "Frozen trial tree")
set(root ${SCENE_TRIAL_ROOT})
add_subdirectory(${root}/source/libsoftgl candidate-library)
add_subdirectory(${root}/baseline-source/libsoftgl baseline-library)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(source ${root}/baseline-source)
        set(library baseline_softgl)
    else()
        set(source ${root}/source)
        set(library softgl)
    endif()
    add_executable(${variant} ${repo}/experiments/glimpsw-mesa-comparison/softgl_bmw.c ${source}/model_wrap.c)
    add_executable(quality_${variant} ${repo}/experiments/scene-hierarchical-depth/quality_frames.c ${source}/model_wrap.c)
    add_executable(resident_${variant} ${repo}/experiments/scene-material-visibility/resident_trial.c ${source}/model_wrap.c)
    foreach(target ${variant} quality_${variant} resident_${variant})
        target_compile_definitions(${target} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
            SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
        target_include_directories(${target} PRIVATE ${source}/libsoftgl/src)
        target_link_libraries(${target} PRIVATE ${library} pthread m)
    endforeach()
    add_executable(quantized_${variant} ${repo}/tests/scene_quantized.c)
    target_include_directories(quantized_${variant} PRIVATE ${source}/libsoftgl/src)
    target_link_libraries(quantized_${variant} PRIVATE ${library} pthread m)
    add_executable(controlled_${variant} controlled.c)
    target_include_directories(controlled_${variant} PRIVATE ${source}/libsoftgl/src ${repo}/tests)
    target_link_libraries(controlled_${variant} PRIVATE ${library} pthread m)
endforeach()
add_executable(position_contract ${repo}/tests/scene_positions.c)
target_include_directories(position_contract PRIVATE ${root}/source/libsoftgl/src)
target_link_libraries(position_contract PRIVATE softgl pthread m)
option(SOFTGL_MSAA_SCALED_AUDIT "Verify actual exact scaled coverage" OFF)
if(SOFTGL_MSAA_SCALED_AUDIT)
    target_compile_definitions(softgl PRIVATE SOFTGL_MSAA_SCALED_AUDIT)
    add_executable(msaa_scaled_contract msaa_scaled_contract.c)
    target_include_directories(msaa_scaled_contract PRIVATE ${root}/source/libsoftgl/src ${root}/source ${repo}/tests)
    target_link_libraries(msaa_scaled_contract PRIVATE softgl pthread m)
    target_compile_definitions(controlled_candidate PRIVATE SOFTGL_MSAA_SCALED_AUDIT)
endif()
