cmake_minimum_required(VERSION 3.20)
project(SceneLazyClusterFrontend LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only
    -msse4.1 -mno-avx -mno-avx2 -mno-avx512f)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCENE_TRIAL_ROOT "${repo}/build/scene-lazy-cluster-frontend/v2" CACHE PATH "Frozen experiment tree")
set(root ${SCENE_TRIAL_ROOT})
add_subdirectory(${root}/source/libsoftgl candidate-library)
add_subdirectory(${root}/baseline-source/libsoftgl baseline-library)
add_subdirectory(${root}/unpruned-source/libsoftgl unpruned-library)
target_compile_definitions(unpruned_softgl PRIVATE SOFTGL_LAZY_CENSUS_ONLY SOFTGL_LAZY_UNBOUNDED SOFTGL_LAZY_AUDIT)
option(SOFTGL_LAZY_AUDIT "Count lazy queries, actual preparation and positions" OFF)
if(SOFTGL_LAZY_AUDIT)
    target_compile_definitions(softgl PRIVATE SOFTGL_LAZY_AUDIT)
endif()
foreach(variant baseline candidate unpruned)
    if(variant STREQUAL "baseline")
        set(source ${root}/baseline-source)
        set(library baseline_softgl)
    elseif(variant STREQUAL "unpruned")
        set(source ${root}/unpruned-source)
        set(library unpruned_softgl)
    else()
        set(source ${root}/source)
        set(library softgl)
    endif()
    foreach(kind resident quality)
        if(kind STREQUAL "resident")
            set(driver ${repo}/experiments/scene-material-visibility/resident_trial.c)
        else()
            set(driver ${repo}/experiments/scene-depth-order-cached-keys/quality_frames.c)
        endif()
        add_executable(${kind}_${variant} ${driver} ${source}/model_wrap.c)
        target_compile_definitions(${kind}_${variant} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
            SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
    foreach(kind quantized small positions hz)
        if(kind STREQUAL "quantized")
            set(fixture ${repo}/tests/scene_quantized.c)
        elseif(kind STREQUAL "small")
            set(fixture ${repo}/experiments/scene-msaa-small-triangles/small_contract.c)
        elseif(kind STREQUAL "positions")
            set(fixture ${repo}/tests/scene_positions.c)
        else()
            set(fixture ${repo}/tests/scene_msaa.c)
        endif()
        add_executable(${kind}_${variant} ${fixture})
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src ${repo}/tests)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
endforeach()

if(SOFTGL_LAZY_AUDIT)
    add_executable(audit_candidate audit_resident.c ${root}/source/model_wrap.c)
    target_compile_definitions(audit_candidate PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
        SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
    target_include_directories(audit_candidate PRIVATE ${root}/source/libsoftgl/src)
    target_link_libraries(audit_candidate PRIVATE softgl pthread m)
endif()

add_executable(audit_unpruned audit_resident.c ${root}/unpruned-source/model_wrap.c)
target_compile_definitions(audit_unpruned PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
    SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
target_include_directories(audit_unpruned PRIVATE ${root}/unpruned-source/libsoftgl/src)
target_link_libraries(audit_unpruned PRIVATE unpruned_softgl pthread m)

foreach(variant candidate unpruned)
    if(variant STREQUAL "candidate")
        set(source ${root}/source)
        set(library softgl)
    else()
        set(source ${root}/unpruned-source)
        set(library unpruned_softgl)
    endif()
    add_executable(lazy_contract_${variant} lazy_contract.c)
    target_compile_definitions(lazy_contract_${variant} PRIVATE SOFTGL_LAZY_AUDIT)
    if(variant STREQUAL "unpruned")
        target_compile_definitions(lazy_contract_${variant} PRIVATE SOFTGL_LAZY_CENSUS_ONLY SOFTGL_LAZY_UNBOUNDED)
    endif()
    target_include_directories(lazy_contract_${variant} PRIVATE ${source}/libsoftgl/src ${repo}/tests)
    target_link_libraries(lazy_contract_${variant} PRIVATE ${library} pthread m)
endforeach()

if(SOFTGL_LAZY_AUDIT AND SOFTGL_LAZY_PROFILE)
    add_executable(profile_candidate profile_resident.c ${root}/source/model_wrap.c)
    target_compile_definitions(profile_candidate PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
        SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
    target_include_directories(profile_candidate PRIVATE ${root}/source/libsoftgl/src)
    target_link_libraries(profile_candidate PRIVATE softgl pthread m)
endif()
