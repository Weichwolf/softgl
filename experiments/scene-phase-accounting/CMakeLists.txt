cmake_minimum_required(VERSION 3.20)
project(ScenePhaseAccounting LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(root ${repo}/build/scene-phase-accounting/source)
add_subdirectory(${root}/libsoftgl library)
add_executable(phase_scene phase_scene.c ${root}/model_wrap.c)
target_compile_definitions(phase_scene PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES
    SOFTGL_MODEL_SCENE_VISIBILITY SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION)
target_include_directories(phase_scene PRIVATE ${root}/libsoftgl/src)
target_compile_options(phase_scene PRIVATE -msse4.1)
target_link_libraries(phase_scene PRIVATE softgl pthread m)
