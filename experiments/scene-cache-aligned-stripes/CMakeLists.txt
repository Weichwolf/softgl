cmake_minimum_required(VERSION 3.20)
project(SceneCacheAlignedStripes LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-O3 -fno-strict-aliasing -ffast-math
    -fno-associative-math -fsigned-zeros -fno-finite-math-only
    -msse4.1 -mno-avx -mno-avx2 -mno-avx512f)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCENE_TRIAL_ROOT "${repo}/build/scene-cache-aligned-stripes" CACHE PATH "Frozen experiment tree")
set(root ${SCENE_TRIAL_ROOT})
add_subdirectory(${root}/source/libsoftgl candidate-library)
add_subdirectory(${root}/baseline-source/libsoftgl baseline-library)
foreach(variant baseline candidate)
    if(variant STREQUAL "baseline")
        set(source ${root}/baseline-source)
        set(library baseline_softgl)
    else()
        set(source ${root}/source)
        set(library softgl)
    endif()
    foreach(kind resident quality)
        if(kind STREQUAL "resident")
            set(driver ${repo}/experiments/scene-material-visibility/resident_trial.c)
        else()
            set(driver ${repo}/experiments/scene-hierarchical-depth/quality_frames.c)
        endif()
        add_executable(${kind}_${variant} ${driver} ${source}/model_wrap.c)
        target_compile_definitions(${kind}_${variant} PRIVATE SOFTGL_MODEL_VERTEX_ATTRIBUTES SOFTGL_MODEL_SCENE_VISIBILITY
            SOFTGL_MODEL_SCENE_POSITIONS SOFTGL_MODEL_TRANSPARENT_FUSION SOFTGL_MODEL_QUANTIZED_VISIBILITY)
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
    foreach(kind quantized small positions hz bins)
        if(kind STREQUAL "quantized")
            set(fixture ${repo}/tests/scene_quantized.c)
        elseif(kind STREQUAL "small")
            set(fixture ${repo}/experiments/scene-msaa-small-triangles/small_contract.c)
        elseif(kind STREQUAL "positions")
            set(fixture ${repo}/tests/scene_positions.c)
        elseif(kind STREQUAL "bins")
            set(fixture ${repo}/tests/scene_bins.c)
        else()
            set(fixture ${repo}/tests/scene_msaa.c)
        endif()
        add_executable(${kind}_${variant} ${fixture})
        target_include_directories(${kind}_${variant} PRIVATE ${source}/libsoftgl/src ${repo}/tests)
        target_link_libraries(${kind}_${variant} PRIVATE ${library} pthread m)
    endforeach()
endforeach()

add_executable(alignment_contract alignment_contract.c)
target_include_directories(alignment_contract PRIVATE ${root}/source/libsoftgl/src)
target_link_libraries(alignment_contract PRIVATE softgl pthread m)
file(STRINGS "${root}/variant.txt" alignment_only REGEX "^stripe_quantum4=1$")
if(alignment_only)
    target_compile_definitions(alignment_contract PRIVATE SOFTGL_STRIPES_ALIGNMENT_ONLY)
endif()
